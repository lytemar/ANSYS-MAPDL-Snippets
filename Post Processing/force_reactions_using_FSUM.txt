!======================================================================================================
! This command snippet extracts force reactions using FSUM
!======================================================================================================


/POST1
ALLSEL
! Set the results step (load step 3, substep 1)
!set,3,1											! For Random Vibration analysis
!set,4,1											! For Response Spectrum analysis
set,last										! Last results set

cmsel,s,Nodal_Named_Selection              		! Select the node set named "Nodal_Named_Selection"
*GET,NumOfNodes,NODE,,COUNT
*del,Node_array
*DIM,Node_array,ARRAY,NumOfNodes
*VGET,Node_array,NODE,,NLIST	                ! Retrieve and store the selected node numbers into the array
*del,VALUE
*DIM,VALUE,,NumOfNodes,3                        ! Define a 2D array to store node location (X) and force reactions (FX, FY)

*DO,I,1,NumOfNodes
    nsel,s,node,,Node_array(I)					! Select (new) a node I from the Node_array
    x_coord = NX(Node_array(I))                 ! Get the X-coordinate of the node
    VALUE(I,1)= x_coord
    ESLN										! Select the elements that are attached to the selected node
    FSUM										! Sum all nodal forces in global coordinate system
    *get,fsumx,fsum,0,item,fx                   ! Retrieve the force sum in the X-direction
    VALUE(I,2)= fsumx
    *get,fsumy,fsum,0,item,fy                   ! Retrieve the force sum in the Y-direction
    VALUE(I,3)= fsumy
*ENDDO

/OUTPUT,JOIN(_wb_userfiles_dir_RS(1),'FILENAME_NEEDED'),csv
*VWRITE, 'Node#', 'LocX(mm)', 'FX(N)', 'FY(N)'
(A13, ',', A13, ',', A13, ',', A13)
*VWRITE,Node_array(1),VALUE(1,1),VALUE(1,2),VALUE(1,3)
(F7.0, ',', F13.2, ',', F13.2, ',', F13.2)

/OUT											! Redirects output to the default location
